cmake_minimum_required(VERSION 3.21)
project(ai_mahjong_lab VERSION 0.1.0 LANGUAGES CXX)
option(AIMJ_BUILD_TESTS "Build unit and integration tests" ON)
option(AIMJ_BUILD_CLI "Build headless benchmark executable" ON)
option(AIMJ_WITH_MNN "Build optional MNN inference adapter" OFF)
option(AIMJ_WITH_SHERPA "Build optional sherpa-onnx TTS adapter" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(WIN32)
  # windows.h defines min/max macros that break std::min/std::max in <algorithm>.
  add_compile_definitions(NOMINMAX)
endif()
find_package(Threads REQUIRED)
add_library(aimj_core STATIC src/game.cpp src/director.cpp src/runtime.cpp src/mock_backends.cpp src/wav.cpp)
target_include_directories(aimj_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(aimj_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(aimj_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(aimj_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AIMJ_WITH_MNN)
  if(NOT DEFINED AIMJ_MNN_INCLUDE_DIR OR NOT DEFINED AIMJ_MNN_LIBRARIES)
    message(FATAL_ERROR "Set AIMJ_MNN_INCLUDE_DIR and AIMJ_MNN_LIBRARIES for pinned MNN build")
  endif()
  target_sources(aimj_core PRIVATE src/mnn_backend.cpp)
  target_compile_definitions(aimj_core PUBLIC AIMJ_WITH_MNN=1)
  target_include_directories(aimj_core PRIVATE "${AIMJ_MNN_INCLUDE_DIR}")
  target_link_libraries(aimj_core PUBLIC ${AIMJ_MNN_LIBRARIES})
endif()
if(AIMJ_WITH_SHERPA)
  if(NOT DEFINED AIMJ_SHERPA_INCLUDE_DIR OR NOT DEFINED AIMJ_SHERPA_LIBRARIES)
    message(FATAL_ERROR "Set AIMJ_SHERPA_INCLUDE_DIR and AIMJ_SHERPA_LIBRARIES for pinned sherpa-onnx build")
  endif()
  target_sources(aimj_core PRIVATE src/sherpa_backend.cpp)
  target_compile_definitions(aimj_core PUBLIC AIMJ_WITH_SHERPA=1)
  target_include_directories(aimj_core PRIVATE "${AIMJ_SHERPA_INCLUDE_DIR}")
  target_link_libraries(aimj_core PUBLIC ${AIMJ_SHERPA_LIBRARIES})
endif()
if(AIMJ_BUILD_CLI)
  add_executable(aimj_bench apps/cli/benchmark.cpp)
  target_link_libraries(aimj_bench PRIVATE aimj_core)
  if(WIN32)
    target_link_libraries(aimj_bench PRIVATE Psapi)
  endif()
endif()
if(AIMJ_BUILD_TESTS)
  enable_testing()
  add_executable(aimj_tests tests/test_main.cpp)
  target_link_libraries(aimj_tests PRIVATE aimj_core)
  add_test(NAME aimj_tests COMMAND aimj_tests)
endif()
