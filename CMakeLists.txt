cmake_minimum_required(VERSION 3.21)
project(ai_mahjong_lab VERSION 0.1.0 LANGUAGES CXX)
option(AIMJ_BUILD_TESTS "Build unit and integration tests" ON)
option(AIMJ_BUILD_CLI "Build headless benchmark executable" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(aimj_core STATIC src/game.cpp src/director.cpp src/runtime.cpp src/mock_backends.cpp src/wav.cpp)
target_include_directories(aimj_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(aimj_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(aimj_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(aimj_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AIMJ_BUILD_CLI)
  add_executable(aimj_bench apps/cli/benchmark.cpp)
  target_link_libraries(aimj_bench PRIVATE aimj_core)
  if(WIN32)
    target_link_libraries(aimj_bench PRIVATE Psapi)
  endif()
endif()
if(AIMJ_BUILD_TESTS)
  enable_testing()
  add_executable(aimj_tests tests/test_main.cpp)
  target_link_libraries(aimj_tests PRIVATE aimj_core)
  add_test(NAME aimj_tests COMMAND aimj_tests)
endif()
